def unique(pairs):
    out = {}
    for k,v in pairs:
        assert k not in out, ('duplicate JSON key',k)
        out[k] = v
    return out


def parse(text):
    return json.loads(text,object_pairs_hook=unique)


def js(path):
    return parse(Path(path).read_text())


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read(path):
    with Path(path).open() as f:
        return list(csv.DictReader(f))


def indexed(path,key='word_id'):
    rows=read(path);out={r[key]:r for r in rows}
    assert len(rows)==len(out), ('duplicate CSV identity',str(path))
    assert all(re.fullmatch(r'[1-9][0-9]*',i) for i in out),str(path)
    return out


def hashes(mapping,base=None):
    for name,digest in mapping.items():
        path=Path(name) if base is None else base/name
        assert sha(path)==digest, ('hash changed',str(path))


def export_manifest(directory):
    m=js(directory/'manifest.json')
    assert m['inputs'] and m['artifacts']
    hashes(m['inputs']);hashes(m.get('implementation',{}));hashes(m['artifacts'],directory)
    return m


def formkey(word):
    return ''.join(c for c in unicodedata.normalize('NFD',word.lower()) if not unicodedata.combining(c))

    for key,name in [('events_sha256','preflight/final-stdout.jsonl'),
                     ('assembled_context_sha256','preflight/final-prompt-input.json'),
                     ('config_sha256','config.json')]:
        assert preflight[key]==sha(G/name),key
    sample=js(SAMPLE/'manifest.json');hashes(sample['outputs'],SAMPLE)

    vocab=indexed(G/'inputs/vocabulary.csv');forms=indexed(SAMPLE/'forms.csv','form_id')
    signal=indexed(SAMPLE/'signals.csv','form_id');erows=indexed(G/'inputs/evidence.csv')
    assert set(vocab)==set(forms)==set(signal)==set(erows) and len(vocab)==1000
    assert sha(G/'inputs/vocabulary.csv')==sha(SAMPLE/'vocabulary.csv')
    assert sha(G/'inputs/evidence.csv')==sha(ROOT/'inputs-final/evidence.csv')

        parents[tuple(r[k] for k in SCOPE)+(r['sense_id'],)].append(r)
    projected=defaultdict(list)

        assert r['word_id']==i and r['word']==vocab[i]['word']

        while ancestor['parent_sense_id']:

            assert key not in seen and len(parents[key])==1,('unresolved parent',i,key)
            seen.add(key);ancestor=parents[key][0]
            assert ancestor['context_status']=='usable_source_bounded'
            texts.append(ancestor['definition'])
        labels=parse(r['tags'])
        assert not (r['number_restriction'] and r['variant_number_restriction'] and r['number_restriction']!=r['variant_number_restriction'])
        expected={'pos':r['pos_detail'] or r['pos'],'definition':r['definition'],
          'number_restriction':r['number_restriction'] or r['variant_number_restriction'],
          'grammar':sorted({x.lower() for t in labels for x in GRAMMAR.findall(t)}),
          'context':list(reversed(texts)),'context_incomplete':False,'usage_labels':sorted(set(labels))}
        assert s==expected,('projection mismatch',i,r['sense_id'])
        assert r['context_text']==' — '.join([*reversed(texts),r['definition']])
        if s not in projected[i]:projected[i].append(s)

        senses=sorted(projected[i],key=lambda s:json.dumps(s,ensure_ascii=False,sort_keys=True))
        assert evidence[i]['senses']==senses,('omitted/invented sense',i)
        positions={re.sub(r'\d.*','',x) for x in parse(forms[i]['loc_pos_json'])}
        assert evidence[i]['valid_pos']==sorted({POS[x] for x in positions}|{s['pos'] for s in senses})

    return vocab,evidence,forms,signal,dict(forms=1000,inventory_rows=1146,with_context=130,
      without_context=870,without_wordfreq=851,without_context_and_wordfreq=764,
      source_contexts=934,projected_contexts=515,slices=dict(slices),protected_files=len(protected))


def check_answer(answer,i,word):
    assert set(answer)=={'word_id','word','level','note'}
    assert type(answer['word_id']) is int and answer['word_id']==int(i) and answer['word']==word
    assert type(answer['level']) is int and 1<=answer['level']<=5
    assert isinstance(answer['note'],str) and answer['note'].strip() and len(answer['note'])<=400


def check_campaign_csv(directory,vocab,records):
    preds=indexed(directory/'predictions.csv');assert set(preds)==set(records)==set(vocab)
    for i,r in records.items():
        assert r['word']==vocab[i]['word'] and r['type']==vocab[i]['type'] and r['word_id']==int(i)
        assert r['confidence'] is None and r['expected_level'] is None and r['probabilities']==[]
        row=preds[i];assert row['status']==r['status'] and row['word']==r['word']
        assert row['rarity_level']==('' if r['rarity_level'] is None else str(r['rarity_level']))
        assert all(row[k]=='' for k in ['confidence','expected_level',*[f'p_L{j}' for j in range(1,6)]])
        assert r['status'] in {'scored','failed'}
        if r['status']=='failed':assert r['rarity_level'] is None and r['error']
        else:assert type(r['rarity_level']) is int and 1<=r['rarity_level']<=5 and not r['error']
    assert read(directory/'run.csv')==[]
    manifest=js(directory/'manifest.json');inventory=[{'word_id':int(i),'word':vocab[i]['word'],'type':vocab[i]['type']} for i in sorted(vocab,key=int)]
    assert manifest['inventory_sha256']==hashlib.sha256(json.dumps(inventory,ensure_ascii=False,sort_keys=True).encode()).hexdigest()
    digest=hashlib.sha256()
    for p in sorted((CODE/'scoring').glob('*.py'))+[CODE/'evaluation/campaign.py',CODE/'evaluation/cli.py']:
        digest.update(str(p.relative_to(CODE)).encode());digest.update(p.read_bytes())
    assert manifest['implementation_sha256']==digest.hexdigest()
    summary=js(directory/'summary.json');assert summary['total']==len(vocab) and summary['pending']==0
    assert summary['scored']==sum(r['status']=='scored' for r in records.values())
    assert summary['failed']==sum(r['status']=='failed' for r in records.values())
    return preds

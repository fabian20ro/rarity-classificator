import csv,json,sys,hashlib
from pathlib import Path

 rows=[{k:r[k] for k in fields} for r in read(source) if ids is None or r['word_id'] in ids]
 _write_csv(target,fields,rows)
project(pilot/'inputs/baseline-oof.csv',root/'baseline.csv',['word_id','word','baseline_level','wordfreq'])
project(pilot/'campaign/predictions.csv',root/'gpt.csv',['word_id','word','status','rarity_level'])

assert actual==[r['word_id'] for r in expected['ranked_queue'][:8]]
assert len(ids)==12 and result['review_eligible']==13
report=dict(status='pass',policy=result['policy'],qwen_route=12,eligible=13,top8=expected['top8_detection'],missed_gpt=['carter'],missed_baseline=['carter','in'],independent_order_match=True,scope='Fixed exploratory policy, reused development labels.

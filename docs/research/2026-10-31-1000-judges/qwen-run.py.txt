class LoggedSession(requests.Session):
 def post(self,url,**kwargs):
  start=time.time();state=json.loads(kwargs['json']['messages'][1]['content'])
  entry=dict(word_id=state['word_id'],started=start,http_status=None)
  try:
   result=super().post(url,**kwargs);entry['http_status']=result.status_code
   try:entry['usage']=result.json().get('usage')
   except ValueError:pass
   return result
  except requests.RequestException as exc:
   entry['transport_error']=str(exc);raise
  finally:
   entry['finished']=time.time()
   with (OUT/'transfers.jsonl').open('a') as f:f.write(json.dumps(entry)+'\n')
